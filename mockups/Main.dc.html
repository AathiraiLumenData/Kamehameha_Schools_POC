WITH latest AS (
  SELECT StdNumber, MAX(SchoolYearStart) AS last_year
  FROM ld_analytics.kschools.student_enrollment
  GROUP BY StdNumber
)
SELECT s.EntrySchoolYearDesc AS entry_year,
       COUNT(*) AS admitted,
       ROUND(100 * AVG(CASE WHEN s.HasGraduated THEN 1 ELSE 0 END), 1) AS pct_graduated,
       ROUND(100 * AVG(CASE WHEN NOT s.HasGraduated
                             AND l.last_year = 2026 THEN 1 ELSE 0 END), 1) AS pct_enrolled
FROM ld_analytics.kschools.student_information s
JOIN latest l USING (StdNumber)
GROUP BY s.EntrySchoolYearDesc
ORDER BY entry_year;
